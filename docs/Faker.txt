import faker
import random
from datetime import timedelta
from django.utils import timezone
from apps.quotations.models import QuotationModel, QuotationItemsModel
from apps.customer.models import CustomerModel
from apps.employee.models import EmployeeModel

fake = faker.Faker('en_US')

def generate_demo_quotations(count=25):
    customers = list(CustomerModel.objects.all())
    employees = list(EmployeeModel.objects.all())

    if not customers or not employees:
        print("Error: You need at least one Customer and one Employee in the DB first!")
        return

    print(f"--- Starting to generate {count} Quotation Records ---")

    for i in range(1, count + 1):
        try:
            customer = random.choice(customers)
            creator = random.choice(employees)
            
            quotation = QuotationModel.objects.create(
                customer_info=customer,
                created_by=creator,
                start_date=timezone.now().date(),
                end_date=timezone.now().date() + timedelta(days=30),
                status=QuotationModel.QuotationStatus.PENDING,
            )

            for _ in range(random.randint(2, 5)):
                QuotationItemsModel.objects.create(
                    base_info=quotation,
                    name=fake.catch_phrase()[:50],
                    quantity=random.randint(1, 10),
                    period=random.choice([1, 3, 6, 12, 24]),
                    price=random.randint(500, 5000)
                )

            quotation.recalculate_total_items()

            print(f"[{i}] Created Quotation: {quotation.code} for {customer.name} (Total: {quotation.total_items})")

        except Exception as e:
            print(f"❌ Error at index {i}: {e}")

    print("\n" + "="*40)
    print(f"FINISH: Successfully generated {count} records!")
    print("="*40)

generate_demo_quotations(25)
